<script>
  // Регистрация Service Worker для Android PWA
  let swReg = null;
  if ('serviceWorker' in navigator) {
    window.addEventListener('load', () => {
      navigator.serviceWorker.register('./sw.js')
        .then(reg => { swReg = reg; })
        .catch(err => console.log('SW registration failed:', err));
    });
  }

  // Звуковой генератор Web Audio API
  function playGong() {
    try {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      const ctx = new AudioCtx();
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();

      osc.type = 'sine';
      osc.frequency.setValueAtTime(440, ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(110, ctx.currentTime + 1.2);

      gain.gain.setValueAtTime(0.4, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 1.2);

      osc.connect(gain);
      gain.connect(ctx.destination);

      osc.start();
      osc.stop(ctx.currentTime + 1.2);
    } catch(e) {}
  }

  // Элементы
  const timeDisplay = document.getElementById('timeDisplay');
  const timerStatusLabel = document.getElementById('timerStatusLabel');
  const circleProgress = document.getElementById('circleProgress');
  const btnStartStop = document.getElementById('btnStartStop');
  const startIcon = document.getElementById('startIcon');
  const startText = document.getElementById('startText');
  const btnReset = document.getElementById('btnReset');
  const presetBtns = document.querySelectorAll('.preset-btn');
  const btnNotifyReq = document.getElementById('btnNotifyReq');
  const breakModal = document.getElementById('breakModal');
  const btnCloseBreak = document.getElementById('btnCloseBreak');

  // Расписание
  const scheduleEnabled = document.getElementById('scheduleEnabled');
  const workStartInput = document.getElementById('workStart');
  const workEndInput = document.getElementById('workEnd');
  const scheduleBadge = document.getElementById('scheduleBadge');

  // Переменные таймера
  const CIRCLE_CIRCUMFERENCE = 2 * Math.PI * 110; // ~691
  circleProgress.style.strokeDasharray = CIRCLE_CIRCUMFERENCE;

  let totalMinutes = 50;
  let remainingSeconds = totalMinutes * 60;
  let timerInterval = null;
  let isRunning = false;

  // Загрузка настроек из localStorage
  if (localStorage.getItem('backTimer_workStart')) workStartInput.value = localStorage.getItem('backTimer_workStart');
  if (localStorage.getItem('backTimer_workEnd')) workEndInput.value = localStorage.getItem('backTimer_workEnd');
  if (localStorage.getItem('backTimer_scheduleEnabled')) {
    scheduleEnabled.checked = localStorage.getItem('backTimer_scheduleEnabled') === 'true';
  }

  // Проверка попадания в рабочий график
  function isWithinWorkSchedule() {
    if (!scheduleEnabled.checked) return true;

    const now = new Date();
    const [startH, startM] = workStartInput.value.split(':').map(Number);
    const [endH, endM] = workEndInput.value.split(':').map(Number);

    const startTotal = startH * 60 + startM;
    const endTotal = endH * 60 + endM;
    const currentTotal = now.getHours() * 60 + now.getMinutes();

    return currentTotal >= startTotal && currentTotal < endTotal;
  }

  function updateScheduleBadge() {
    if (!scheduleEnabled.checked) {
      scheduleBadge.textContent = 'Всегда активен';
      scheduleBadge.className = 'badge-status badge-in-work';
      return;
    }
    if (isWithinWorkSchedule()) {
      scheduleBadge.textContent = 'В графике';
      scheduleBadge.className = 'badge-status badge-in-work';
    } else {
      scheduleBadge.textContent = 'Вне графика';
      scheduleBadge.className = 'badge-status badge-off-work';
    }
  }

  // Обновление интерфейса
  function renderTime() {
    const mins = Math.floor(remainingSeconds / 60);
    const secs = remainingSeconds % 60;
    timeDisplay.textContent = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
    document.title = `${timeDisplay.textContent} — Спина`;

    const totalSeconds = totalMinutes * 60;
    const offset = CIRCLE_CIRCUMFERENCE * (1 - remainingSeconds / totalSeconds);
    circleProgress.style.strokeDashoffset = offset;
  }

  // Запуск / Пауза
  function toggleTimer() {
    if (isRunning) {
      pauseTimer();
    } else {
      startTimer();
    }
  }

  function startTimer() {
    if (!isWithinWorkSchedule()) {
      if (!confirm("Сейчас время вне установленного рабочего графика. Все равно запустить таймер?")) {
        return;
      }
    }
    isRunning = true;
    startText.textContent = "Пауза";
    startIcon.textContent = "⏸";
    timerStatusLabel.textContent = "Концентрация";

    timerInterval = setInterval(() => {
      if (remainingSeconds > 0) {
        remainingSeconds--;
        renderTime();
      } else {
        completeTimer();
      }
    }, 1000);
  }

  function pauseTimer() {
    isRunning = false;
    clearInterval(timerInterval);
    startText.textContent = "Старт";
    startIcon.textContent = "▶";
    timerStatusLabel.textContent = "На паузе";
  }

  function resetTimer() {
    pauseTimer();
    remainingSeconds = totalMinutes * 60;
    timerStatusLabel.textContent = "Готов";
    renderTime();
  }

  // Окончание интервала
  function completeTimer() {
    pauseTimer();
    playGong();

    // Системное уведомление Android
    if (Notification.permission === 'granted') {
      const options = {
        body: '50 минут истекли! Сделайте декомпрессию поясницы и выпейте воды.',
        icon: 'data:image/svg+xml,%3Csvg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 100 100"%3E%3Ccircle cx="50" cy="50" r="45" fill="%23B82429"/%3E%3C/svg%3E',
        vibrate: [200, 100, 200, 100, 200],
        requireInteraction: true
      };
      if (swReg && swReg.showNotification) {
        swReg.showNotification('Пора разгрузить спину!', options);
      } else {
        new Notification('Пора разгрузить спину!', options);
      }
    }

    breakModal.classList.add('active');
  }

  // Слушатели событий
  btnStartStop.addEventListener('click', toggleTimer);
  btnReset.addEventListener('click', resetTimer);

  presetBtns.forEach(btn => {
    btn.addEventListener('click', () => {
      presetBtns.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      totalMinutes = parseInt(btn.dataset.time, 10);
      resetTimer();
    });
  });

  btnCloseBreak.addEventListener('click', () => {
    breakModal.classList.remove('active');
    resetTimer();
    startTimer();
  });

  btnNotifyReq.addEventListener('click', async () => {
    if ('Notification' in window) {
      const perm = await Notification.requestPermission();
      if (perm === 'granted') {
        alert('Уведомления успешно подключены!');
      }
    }
  });

  // Автосохранение графика
  [workStartInput, workEndInput, scheduleEnabled].forEach(elem => {
    elem.addEventListener('change', () => {
      localStorage.setItem('backTimer_workStart', workStartInput.value);
      localStorage.setItem('backTimer_workEnd', workEndInput.value);
      localStorage.setItem('backTimer_scheduleEnabled', scheduleEnabled.checked);
      updateScheduleBadge();
    });
  });

  // Проверка графика каждые 30 секунд
  setInterval(updateScheduleBadge, 30000);
  updateScheduleBadge();
  renderTime();
</script>
